    <!-- Firebase SDK & Logic -->
    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";
        import { getDatabase, ref, onValue, push, remove } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-database.js";

        const firebaseConfig = {
            apiKey: "AIzaSyD5BLPlTjaDBy6hmPeuedHtCnmcBQozLC4",
            authDomain: "khan-petroleum-service-abuja.firebaseapp.com",
            projectId: "khan-petroleum-service-abuja",
            storageBucket: "khan-petroleum-service-abuja.firebasestorage.app",
            messagingSenderId: "337162512372",
            appId: "1:337162512372:web:b6f0f2be3a6d0ee7163ba1",
            measurementId: "G-GVB8XSJNJX"
        };

        const app = initializeApp(firebaseConfig);
        const db = getDatabase(app);

        document.getElementById('inputDate').value = new Date().toISOString().split('T')[0];

        // Load Customers Dropdown
        onValue(ref(db, 'customers'), (snapshot) => {
            const data = snapshot.val();
            const custSelect = document.getElementById('inputCustomer');
            custSelect.innerHTML = '<option value="" selected disabled>-- Select Customer --</option>';
            if (data) {
                Object.keys(data).forEach(k => {
                    const c = data[k];
                    const opt = document.createElement('option');
                    opt.value = `[${c.code || ''}] ${c.name}`;
                    opt.textContent = `[${c.code || ''}] ${c.name}`;
                    custSelect.appendChild(opt);
                });
            }
        });

        // Load Bank Accounts & Cash in Hand Dropdown
        onValue(ref(db, 'bank_accounts'), (snapshot) => {
            const data = snapshot.val();
            const accSelect = document.getElementById('inputAccount');
            accSelect.innerHTML = '<option value="" selected disabled>-- Account --</option>';
            
            const cashOpt = document.createElement('option');
            cashOpt.value = "Cash in Hand";
            cashOpt.textContent = "Cash in Hand";
            accSelect.appendChild(cashOpt);

            if (data) {
                Object.keys(data).forEach(k => {
                    const b = data[k];
                    const opt = document.createElement('option');
                    opt.value = `${b.title} (${b.bankName})`;
                    opt.textContent = `${b.title} (${b.bankName})`;
                    accSelect.appendChild(opt);
                });
            }
        });

        // Add Button Click
        function handleAdd() {
            const date = document.getElementById('inputDate').value;
            const customer = document.getElementById('inputCustomer').value;
            const amount = parseFloat(document.getElementById('inputAmount').value) || 0;
            const method = document.getElementById('inputMethod').value;
            const account = document.getElementById('inputAccount').value;
            const description = document.getElementById('inputDesc').value || 'Payment';

            if (!customer || amount <= 0 || !account) {
                alert("Baraye karam Customer, Amount aur Account zaroor select/enter karein!");
                return;
            }

            const newEntry = { date, customer, amount, method, account, description };

            push(ref(db, 'payments'), newEntry).then(() => {
                document.getElementById('inputDesc').value = '';
                document.getElementById('inputAmount').value = '';
                document.getElementById('inputDate').value = new Date().toISOString().split('T')[0];
            });
        }

        document.getElementById('addBtn').addEventListener('click', handleAdd);

        // Load Payments Table & keep input row at top
        onValue(ref(db, 'payments'), (snapshot) => {
            const data = snapshot.val();
            const tbody = document.getElementById('paymentTableBody');
            
            // Remove existing data rows (keep only input row)
            while (tbody.rows.length > 1) {
                tbody.deleteRow(1);
            }

            if (data) {
                let index = 1;
                Object.keys(data).forEach(k => {
                    const item = data[k];
                    const tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td class="text-center fw-bold">${index++}</td>
                        <td class="text-center">${item.date || '-'}</td>
                        <td>${item.description || '-'}</td>
                        <td>${item.customer || '-'}</td>
                        <td class="text-end text-danger fw-semibold">${item.amount ? item.amount.toLocaleString() : '0'}</td>
                        <td class="text-center">${item.method || '-'}</td>
                        <td>${item.account || '-'}</td>
                        <td class="text-center">
                            <button class="btn btn-danger btn-sm py-0 px-2 delete-btn" data-id="${k}"><i class="fa-solid fa-trash"></i></button>
                        </td>
                    `;
                    tbody.appendChild(tr);
                });

                // Delete buttons event listener
                document.querySelectorAll('.delete-btn').forEach(btn => {
                    btn.addEventListener('click', (e) => {
                        const id = e.currentTarget.getAttribute('data-id');
                        if (confirm("Kya aap is entry ko delete karna chahte hain?")) {
                            remove(ref(db, `payments/${id}`));
                        }
                    });
                });
            }
        });
